## Part 4: Real-time Price Calculation

┌─────────────────────────────────────────┐
│   participant-payment.js                │
│                                         │
│   ticketsCountInput.addEventListener(   │
│     "input",                            │
│     updatePaymentSummary                │
│   );                                    │
│                                         │
│   function updatePaymentSummary() {     │
│     const ticketsCount =                │
│       parseInt(                         │
│         ticketsCountInput.value         │
│       ) || 1;                           │
│                                         │
│     const pricePerTicket =              │
│       parseFloat(                       │
│         currentEvent.price              │
│       ) || 0;                           │
│                                         │
│     const totalAmount =                 │
│       pricePerTicket * ticketsCount;    │
│                                         │
│     pricePerTicketDisplay.textContent = │
│       `₹ ${pricePerTicket.toFixed(2)}`; │
│                                         │
│     ticketsCountDisplay.textContent =   │
│       ticketsCount;                     │
│                                         │
│     totalAmountDisplay.textContent =    │
│       `₹ ${totalAmount.toFixed(2)}`;    │
│   }                                     │
└─────────────────────────────────────────┘

User changes tickets from 1 → 2:
  Price per ticket: ₹ 499.00 (stays same)
  Number of tickets: 1 → 2
  Total Amount: ₹ 499.00 → ₹ 998.00 ✨

┌─────────────────────────────────────────┐
│   Backend: routes/payments.js           │
│                                         │
│   router.post('/', async (req, res) => {│
│     const {                             │
│       booking_id,                       │
│       participant_name,                 │
│       event_id,                         │
│       event_title,                      │
│       tickets_count,                    │
│       total_amount,                     │
│       payment_method                    │
│     } = req.body;                       │
│                                         │
│     // Validate price                   │
│     const event = await pool.query(     │
│       `SELECT price FROM events        │
│        WHERE event_id = $1`,           │
│       [event_id]                        │
│     );                                  │
│                                         │
│     const expectedAmount =              │
│       event.price * tickets_count;      │
│                                         │
│     if (Math.abs(expectedAmount -       │
│         total_amount) > 0.01) {         │
│       return res.status(400).json({     │
│         message: "Price mismatch"       │
│       });                               │
│     }                                   │
│                                         │
│     // Generate transaction ID          │
│     const transaction_id =              │
│       `TXN${Date.now()}${Math.floor(   │
│         Math.random() * 1000           │
│       )}`;                              │
│                                         │
│     // Insert payment                   │
│     const result = await pool.query(    │
│       `INSERT INTO payments            │
│        (booking_id, participant_name,  │
│         event_id, event_title,         │
│         tickets_count, total_amount,   │
│         payment_method,                │
│         payment_status,                │
│         transaction_id)                │
│        VALUES ($1,$2,$3,$4,$5,$6,      │
│                $7,'Paid',$8)           │
│        RETURNING *`,                    │
│       [booking_id, participant_name,    │
│        event_id, event_title,          │
│        tickets_count, total_amount,    │
│        payment_method, transaction_id] │
│     );                                  │
│                                         │
│     res.status(201).json({              │
│       success: true,                    │
│       message: "Payment successful!",   │
│       payment: result.rows[0]           │
│     });                                 │
│   });                                   │
